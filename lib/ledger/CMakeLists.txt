# interface.json is the library interface's description, which describe serves: it is compiled in as
# bytes, so the file stays the one source of it.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/interface.json" description_hex HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," description_bytes "${description_hex}")
set(description_c "${CMAKE_CURRENT_BINARY_DIR}/generated/interface_description.c")
set(description_text "/* Generated from interface.json by CMakeLists.txt. */\n#include <stddef.h>\nconst char ll_interface_description[] = {${description_bytes}0};\nconst size_t ll_interface_description_len = sizeof ll_interface_description - 1;\n")
if(EXISTS "${description_c}")
  file(READ "${description_c}" description_old)
endif()
if(NOT description_old STREQUAL description_text)
  file(WRITE "${description_c}" "${description_text}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS interface.json)
list(APPEND ledger_sources "${description_c}")

function(ledger_target name kind)
  add_library(${name} ${kind} ${ledger_sources})
  target_include_directories(${name} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_compile_options(${name} PRIVATE ${LEDGER_WARN})
  target_compile_definitions(${name} PRIVATE EXT_LEDGER_BUILD)
  set_target_properties(${name} PROPERTIES C_VISIBILITY_PRESET hidden)
  target_link_libraries(${name} PRIVATE Threads::Threads)
  if(EXT_LEDGER_PORTABLE)
    # No native frameworks: only the platform store primitives.
  elseif(WIN32)
    target_link_libraries(${name} PRIVATE advapi32 user32 psapi iphlpapi ws2_32) # process, memory and socket inventory
  elseif(APPLE)
    target_link_libraries(${name} PRIVATE "-framework CoreFoundation") # bundle metadata
    if(LEDGER_APPLE_EMBEDDED)
      target_link_libraries(${name} PRIVATE "-framework UIKit") # iOS battery state
    endif()
    if(NOT LEDGER_APPLE_EMBEDDED)
      target_link_libraries(${name} PRIVATE "-framework ApplicationServices") # CoreGraphics and the Accessibility API
    endif()
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(${name} PRIVATE ${CMAKE_DL_LIBS}) # libX11 is loaded when it is there, never linked
  endif()
  if(NOT WIN32 AND NOT APPLE)
    target_link_libraries(${name} PRIVATE m)
  endif()
  if(EXT_LEDGER_SQLITE)
    target_link_libraries(${name} PRIVATE ${LEDGER_SQLITE_TARGET})
    target_compile_definitions(${name} PRIVATE EXT_LEDGER_SQLITE)
  endif()
endfunction()
