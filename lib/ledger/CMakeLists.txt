if(MSVC)

# iOS, tvOS, watchOS and visionOS have no shell to run a command from, and no
# ApplicationServices.
if(CMAKE_SYSTEM_NAME MATCHES "^(iOS|tvOS|watchOS|visionOS)$")
  set(LEDGER_APPLE_EMBEDDED TRUE)
  set(ledger_command_default OFF)
else()
  set(LEDGER_APPLE_EMBEDDED FALSE)
  set(ledger_command_default ON)
endif()

find_package(Threads REQUIRED)
include(GNUInstallDirs)

function(ledger_target name kind)
  add_library(${name} ${kind} ${ledger_sources})
  target_include_directories(${name} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_compile_options(${name} PRIVATE ${LEDGER_WARN})
  target_compile_definitions(${name} PRIVATE EXT_LEDGER_BUILD)
  set_target_properties(${name} PROPERTIES C_VISIBILITY_PRESET hidden)
  target_link_libraries(${name} PRIVATE Threads::Threads)
  if(WIN32)
    target_link_libraries(${name} PRIVATE advapi32 user32) # process tokens, SIDs, the registry; displays and windows
  elseif(APPLE)
    target_link_libraries(${name} PRIVATE "-framework CoreFoundation") # bundle metadata
    if(NOT LEDGER_APPLE_EMBEDDED)
      target_link_libraries(${name} PRIVATE "-framework ApplicationServices") # CoreGraphics and the Accessibility API
    endif()
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(${name} PRIVATE ${CMAKE_DL_LIBS}) # libX11 is loaded when it is there, never linked
  endif()
  if(NOT WIN32 AND NOT APPLE)
    target_link_libraries(${name} PRIVATE m)
  endif()
  if(EXT_LEDGER_SQLITE)
    target_link_libraries(${name} PRIVATE ${LEDGER_SQLITE_TARGET})
    target_compile_definitions(${name} PRIVATE EXT_LEDGER_SQLITE)
  endif()
endfunction()

ledger_target(ext_ledger_static STATIC)
target_compile_definitions(ext_ledger_static PUBLIC EXT_LEDGER_STATIC)
if(MSVC)
  # MSVC names a static library and a DLL's import library alike (ext_ledger.lib).
  set_target_properties(ext_ledger_static PROPERTIES OUTPUT_NAME ext_ledger_static)
else()
  set_target_properties(ext_ledger_static PROPERTIES OUTPUT_NAME ext_ledger)
endif()
add_library(EXT::ledger_static ALIAS ext_ledger_static)
set(ledger_targets ext_ledger_static)
if(EXT_LEDGER_SHARED)
  ledger_target(ext_ledger_shared SHARED)
  set_target_properties(ext_ledger_shared PROPERTIES OUTPUT_NAME ext_ledger SOVERSION 0)
  add_library(EXT::ledger_shared ALIAS ext_ledger_shared)
  list(APPEND ledger_targets ext_ledger_shared)
endif()

option(EXT_LEDGER_COMMAND "Build the ext-ledger command" ${ledger_command_default})
if(EXT_LEDGER_COMMAND)
  add_executable(ext-ledger cmd/ext-ledger.c)
  target_compile_options(ext-ledger PRIVATE ${LEDGER_WARN})
  target_link_libraries(ext-ledger PRIVATE ext_ledger_static Threads::Threads)
  install(TARGETS ext-ledger RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

install(TARGETS ${ledger_targets} EXPORT ext_ledger_targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
