include(GNUInstallDirs)
if(MSVC)
  # cl takes /W3 and /FI where gcc and clang take -Wall -Wextra -Werror and -include.
  set(EXT_WARN /W3 /utf-8)
  set(EXT_TEST_WARN /W3 /utf-8 /UNDEBUG)
  set(EXT_FORCE_INCLUDE /FI)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
  add_compile_options(/experimental:c11atomics)
else()
  set(EXT_WARN -Wall -Wextra -Werror)
  set(EXT_TEST_WARN -Wall -Wextra -Werror -UNDEBUG)
  set(EXT_FORCE_INCLUDE -include)
endif()
# Rename every external parser function in both declarations and definitions.
# The vendored source stays unmodified. Regenerate on parser upgrades.
file(STRINGS vendor/yyjson.h parser_declarations REGEX "^yyjson_api ")
set(parser_namespace "/* Generated private parser namespace; inherit hidden visibility. */\n#define yyjson_api\n")
foreach(declaration IN LISTS parser_declarations)
  string(REGEX MATCH "[A-Za-z_][A-Za-z_0-9]*\\(" symbol "${declaration}")
  if(NOT symbol)
    message(FATAL_ERROR "Unrecognized yyjson declaration: ${declaration}")
  endif()
  string(REPLACE "(" "" symbol "${symbol}")
  string(APPEND parser_namespace "#define ${symbol} ext_host_private_${symbol}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/parser_namespace.h" "${parser_namespace}")

set(core_sources src/common.c src/wire.c src/progress.c src/reverse.c src/endpoint.c src/services.c src/guest.c src/sha256.c src/packages.c vendor/yyjson.c)
add_library(ext_guest_static STATIC ${core_sources})
set_target_properties(ext_guest_static PROPERTIES EXPORT_NAME guest_static)
target_compile_definitions(ext_guest_static PUBLIC EXT_HOST_STATIC)
if(WIN32)
  # clock_gettime comes from winpthreads on some Windows toolchains (clang on
  # ARM64), so even the guest-only library needs it at link time.
  find_package(Threads)
  if(Threads_FOUND)
    target_link_libraries(ext_guest_static PUBLIC "$<BUILD_INTERFACE:Threads::Threads>")
  endif()
endif()
add_library(ext_guest INTERFACE)
target_link_libraries(ext_guest INTERFACE ext_guest_static)
set_target_properties(ext_guest PROPERTIES EXPORT_NAME guest)
add_library(EXT::guest ALIAS ext_guest)
add_library(EXT::guest_static ALIAS ext_guest_static)
set(engine_targets ext_guest_static)
if(EXT_BUILD_SHARED)
  add_library(ext_guest_shared SHARED ${core_sources})
  set_target_properties(ext_guest_shared PROPERTIES OUTPUT_NAME ext_guest EXPORT_NAME guest_shared SOVERSION 2)
  add_library(EXT::guest_shared ALIAS ext_guest_shared)
  list(APPEND engine_targets ext_guest_shared)
endif()
set(interface_targets ext_guest)
if(EXT_BUILD_HOST)
  if(WIN32)
    set(platform_sources src/channel_win32.c src/fs_win32.c src/random_win32.c)
  else()
    set(platform_sources src/channel_posix.c src/fs_posix.c src/fs_write_posix.c src/random_posix.c)
  endif()
  set(host_sources ${core_sources} src/host.c ${platform_sources} src/integrity.c src/store.c src/trust.c src/instances.c src/streams.c)
  add_library(ext_host_static STATIC ${host_sources})
  set_target_properties(ext_host_static PROPERTIES EXPORT_NAME host_static)
  target_compile_definitions(ext_host_static PUBLIC EXT_HOST_STATIC)
  list(APPEND engine_targets ext_host_static)
  if(EXT_BUILD_SHARED)
    add_library(ext_host_shared SHARED ${host_sources})
    set_target_properties(ext_host_shared PROPERTIES OUTPUT_NAME ext_host EXPORT_NAME host_shared SOVERSION 2)
    list(APPEND engine_targets ext_host_shared)
  endif()
endif()
foreach(target IN LISTS engine_targets)
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src vendor)
  if(NOT MSVC)
    target_link_libraries(${target} PRIVATE m)
  endif()
  if(target STREQUAL "ext_host_static" OR target STREQUAL "ext_host_shared")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(WIN32)
      target_link_libraries(${target} PRIVATE bcrypt)
    endif()
  endif()
  set_target_properties(${target} PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
  target_compile_definitions(${target} PRIVATE EXT_HOST_BUILD)
  target_compile_options(${target} PRIVATE ${EXT_WARN}
    ${EXT_FORCE_INCLUDE} "${CMAKE_CURRENT_BINARY_DIR}/parser_namespace.h")
  if(EXT_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  elseif(EXT_THREAD_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=thread)
  elseif(EXT_FUZZ)
    target_compile_options(${target} PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
