cmake_minimum_required(VERSION 3.20)
project(ext_system_net LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "pkg/ext-linux/net requires Linux")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(ext_system_net SHARED src/network.c)
target_include_directories(ext_system_net PUBLIC include)
target_compile_definitions(ext_system_net PRIVATE _GNU_SOURCE)
target_compile_options(ext_system_net PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(ext_system_net_test tests/network.c)
  target_link_libraries(ext_system_net_test PRIVATE ext_system_net)
  add_test(NAME ext_system_net_test COMMAND ext_system_net_test)
  set(NETWORK_FIXTURES ${CMAKE_CURRENT_SOURCE_DIR}/../../../res/system/testdata/network/v1/cases.json)
  if(EXISTS ${NETWORK_FIXTURES})
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME ext_system_net_fixtures
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures.py
              $<TARGET_FILE:ext_system_net> ${NETWORK_FIXTURES})
  endif()
endif()
