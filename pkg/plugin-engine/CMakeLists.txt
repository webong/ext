cmake_minimum_required(VERSION 3.18)
project(ctx_host_prototype VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(CTX_BUILD_HOST "Build native host, filesystem and resource runtimes" ON)
if(CTX_BUILD_HOST AND NOT UNIX)
  message(FATAL_ERROR "Native host currently supports Linux/macOS; use CTX_BUILD_HOST=OFF for the portable guest core")
endif()
option(CTX_BUILD_SHARED "Also build the optional shared engine" OFF)
option(CTX_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CTX_THREAD_SANITIZE "Enable the C thread sanitizer" OFF)
if(CTX_SANITIZE AND CTX_THREAD_SANITIZE)
  message(FATAL_ERROR "Address and thread sanitizers require separate builds")
endif()
if(CTX_BUILD_HOST)
  find_package(Threads REQUIRED)
endif()
include(GNUInstallDirs)
# Rename every external parser function in both declarations and definitions.
# The vendored source stays unmodified. Regenerate on parser upgrades.
file(STRINGS vendor/yyjson.h parser_declarations REGEX "^yyjson_api ")
set(parser_namespace "/* Generated private parser namespace; inherit hidden visibility. */\n#define yyjson_api\n")
foreach(declaration IN LISTS parser_declarations)
  string(REGEX MATCH "[A-Za-z_][A-Za-z_0-9]*\\(" symbol "${declaration}")
  if(NOT symbol)
    message(FATAL_ERROR "Unrecognized yyjson declaration: ${declaration}")
  endif()
  string(REPLACE "(" "" symbol "${symbol}")
  string(APPEND parser_namespace "#define ${symbol} ctx_host_private_${symbol}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/parser_namespace.h" "${parser_namespace}")

set(core_sources src/common.c src/wire.c src/services.c src/guest.c src/sha256.c src/packages.c vendor/yyjson.c)
add_library(ctx_guest_static STATIC ${core_sources})
set_target_properties(ctx_guest_static PROPERTIES EXPORT_NAME guest_static)
target_compile_definitions(ctx_guest_static PUBLIC CTX_HOST_STATIC)
add_library(ctx_guest INTERFACE)
target_link_libraries(ctx_guest INTERFACE ctx_guest_static)
set_target_properties(ctx_guest PROPERTIES EXPORT_NAME guest)
add_library(CTX::guest ALIAS ctx_guest)
add_library(CTX::guest_static ALIAS ctx_guest_static)
set(engine_targets ctx_guest_static)
if(CTX_BUILD_SHARED)
  add_library(ctx_guest_shared SHARED ${core_sources})
  set_target_properties(ctx_guest_shared PROPERTIES OUTPUT_NAME ctx_guest EXPORT_NAME guest_shared SOVERSION 2)
  add_library(CTX::guest_shared ALIAS ctx_guest_shared)
  list(APPEND engine_targets ctx_guest_shared)
endif()
set(interface_targets ctx_guest)
if(CTX_BUILD_HOST)
  set(host_sources ${core_sources} src/host.c src/integrity.c src/instances.c src/streams.c)
  add_library(ctx_host_static STATIC ${host_sources})
  set_target_properties(ctx_host_static PROPERTIES EXPORT_NAME host_static)
  target_compile_definitions(ctx_host_static PUBLIC CTX_HOST_STATIC)
  list(APPEND engine_targets ctx_host_static)
  if(CTX_BUILD_SHARED)
    add_library(ctx_host_shared SHARED ${host_sources})
    set_target_properties(ctx_host_shared PROPERTIES OUTPUT_NAME ctx_host EXPORT_NAME host_shared SOVERSION 2)
    list(APPEND engine_targets ctx_host_shared)
  endif()
endif()
foreach(target IN LISTS engine_targets)
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src vendor)
  target_link_libraries(${target} PRIVATE m)
  if(target STREQUAL "ctx_host_static" OR target STREQUAL "ctx_host_shared")
    target_link_libraries(${target} PRIVATE Threads::Threads)
  endif()
  set_target_properties(${target} PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
  target_compile_definitions(${target} PRIVATE CTX_HOST_BUILD)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
    -include "${CMAKE_CURRENT_BINARY_DIR}/parser_namespace.h")
  if(CTX_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  elseif(CTX_THREAD_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=thread)
  endif()
endforeach()
add_executable(ctx_guest_core_test tests/guest_core.c)
target_link_libraries(ctx_guest_core_test PRIVATE ctx_guest)
target_compile_options(ctx_guest_core_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(CTX_SANITIZE)
  target_compile_options(ctx_guest_core_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
elseif(CTX_THREAD_SANITIZE)
  target_compile_options(ctx_guest_core_test PRIVATE -fsanitize=thread)
endif()
if(CTX_BUILD_HOST)
# The default embedding choice is static; execution backend is a separate choice.
add_library(ctx_host INTERFACE)
target_link_libraries(ctx_host INTERFACE ctx_host_static)
set_target_properties(ctx_host PROPERTIES EXPORT_NAME host)
list(APPEND interface_targets ctx_host)
add_library(CTX::host ALIAS ctx_host)
add_library(CTX::host_static ALIAS ctx_host_static)
if(CTX_BUILD_SHARED)
  add_library(CTX::host_shared ALIAS ctx_host_shared)
endif()
add_executable(ctx_host_native_test tests/native.c)
add_executable(ctx_resources_test tests/resources.c)
foreach(test_target IN ITEMS ctx_host_native_test ctx_resources_test)
  target_link_libraries(${test_target} PRIVATE ctx_host Threads::Threads)
  target_compile_options(${test_target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  if(CTX_SANITIZE)
    target_compile_options(${test_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(CTX_THREAD_SANITIZE)
    target_compile_options(${test_target} PRIVATE -fsanitize=thread)
  endif()
endforeach()
endif()
