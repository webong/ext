cmake_minimum_required(VERSION 3.18)
project(ext_host_prototype VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(EXT_BUILD_HOST "Build native host, filesystem and resource runtimes" ON)
if(EXT_BUILD_HOST AND NOT UNIX)
  message(FATAL_ERROR "Native host currently supports Linux/macOS; use EXT_BUILD_HOST=OFF for the portable guest core")
endif()
option(EXT_BUILD_SHARED "Also build the optional shared engine" OFF)
option(EXT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(EXT_THREAD_SANITIZE "Enable the C thread sanitizer" OFF)
if(EXT_SANITIZE AND EXT_THREAD_SANITIZE)
  message(FATAL_ERROR "Address and thread sanitizers require separate builds")
endif()
if(EXT_BUILD_HOST)
  find_package(Threads REQUIRED)
endif()
include(GNUInstallDirs)
# Rename every external parser function in both declarations and definitions.
# The vendored source stays unmodified. Regenerate on parser upgrades.
file(STRINGS vendor/yyjson.h parser_declarations REGEX "^yyjson_api ")
set(parser_namespace "/* Generated private parser namespace; inherit hidden visibility. */\n#define yyjson_api\n")
foreach(declaration IN LISTS parser_declarations)
  string(REGEX MATCH "[A-Za-z_][A-Za-z_0-9]*\\(" symbol "${declaration}")
  if(NOT symbol)
    message(FATAL_ERROR "Unrecognized yyjson declaration: ${declaration}")
  endif()
  string(REPLACE "(" "" symbol "${symbol}")
  string(APPEND parser_namespace "#define ${symbol} ext_host_private_${symbol}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/parser_namespace.h" "${parser_namespace}")

set(core_sources src/common.c src/wire.c src/services.c src/guest.c src/sha256.c src/packages.c vendor/yyjson.c)
add_library(ext_guest_static STATIC ${core_sources})
set_target_properties(ext_guest_static PROPERTIES EXPORT_NAME guest_static)
target_compile_definitions(ext_guest_static PUBLIC EXT_HOST_STATIC)
add_library(ext_guest INTERFACE)
target_link_libraries(ext_guest INTERFACE ext_guest_static)
set_target_properties(ext_guest PROPERTIES EXPORT_NAME guest)
add_library(EXT::guest ALIAS ext_guest)
add_library(EXT::guest_static ALIAS ext_guest_static)
set(engine_targets ext_guest_static)
if(EXT_BUILD_SHARED)
  add_library(ext_guest_shared SHARED ${core_sources})
  set_target_properties(ext_guest_shared PROPERTIES OUTPUT_NAME ext_guest EXPORT_NAME guest_shared SOVERSION 2)
  add_library(EXT::guest_shared ALIAS ext_guest_shared)
  list(APPEND engine_targets ext_guest_shared)
endif()
set(interface_targets ext_guest)
if(EXT_BUILD_HOST)
  set(host_sources ${core_sources} src/host.c src/integrity.c src/instances.c src/streams.c)
  add_library(ext_host_static STATIC ${host_sources})
  set_target_properties(ext_host_static PROPERTIES EXPORT_NAME host_static)
  target_compile_definitions(ext_host_static PUBLIC EXT_HOST_STATIC)
  list(APPEND engine_targets ext_host_static)
  if(EXT_BUILD_SHARED)
    add_library(ext_host_shared SHARED ${host_sources})
    set_target_properties(ext_host_shared PROPERTIES OUTPUT_NAME ext_host EXPORT_NAME host_shared SOVERSION 2)
    list(APPEND engine_targets ext_host_shared)
  endif()
endif()
foreach(target IN LISTS engine_targets)
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src vendor)
  target_link_libraries(${target} PRIVATE m)
  if(target STREQUAL "ext_host_static" OR target STREQUAL "ext_host_shared")
    target_link_libraries(${target} PRIVATE Threads::Threads)
  endif()
  set_target_properties(${target} PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
  target_compile_definitions(${target} PRIVATE EXT_HOST_BUILD)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
    -include "${CMAKE_CURRENT_BINARY_DIR}/parser_namespace.h")
  if(EXT_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  elseif(EXT_THREAD_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=thread)
  endif()
endforeach()

target_link_libraries(ext_guest_core_test PRIVATE ext_guest)
target_compile_options(ext_guest_core_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(EXT_SANITIZE)
  target_compile_options(ext_guest_core_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
elseif(EXT_THREAD_SANITIZE)
  target_compile_options(ext_guest_core_test PRIVATE -fsanitize=thread)
endif()
if(EXT_BUILD_HOST)
# The default embedding choice is static; execution backend is a separate choice.

target_link_libraries(ext_host INTERFACE ext_host_static)

list(APPEND interface_targets ext_host)
add_library(EXT::host ALIAS ext_host)
add_library(EXT::host_static ALIAS ext_host_static)
if(EXT_BUILD_SHARED)
  add_library(EXT::host_shared ALIAS ext_host_shared)
endif()

add_executable(ext_resources_test tests/resources.c)
foreach(test_target IN ITEMS ext_host_native_test ext_resources_test)
  target_link_libraries(${test_target} PRIVATE ext_host Threads::Threads)
  target_compile_options(${test_target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  if(EXT_SANITIZE)
    target_compile_options(${test_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(EXT_THREAD_SANITIZE)
    target_compile_options(${test_target} PRIVATE -fsanitize=thread)
  endif()
endforeach()
endif()
